- [ ] **Step 1: Write the Kerr geodesic helper functions**

```glsl
// Add at top of FRAG_SCENE, after constants (line 228):
const float M = 0.5;           // black hole mass (geometrized)
const float ESC = 25.0;        // escape radius

// Kerr metric functions
float Sigma(float r, float th) { return r*r + a*a*cos(th)*cos(th); }
float Delta(float r) { return r*r - 2.0*M*r + a*a; }

// First-order Kerr geodesic RHS: d/dλ (r, th, phi, t) = f(r, th, phi, t; E, Lz, Q)
// Returns vec4(dr, dth, dphi, dt)
vec4 kerr_rhs(float r, float th, float E, float Lz, float Q) {
  float a = u_spin * M;  // dimensionful spin
  float S = Sigma(r, th);
  float D = Delta(r);
  float P = E*(r*r + a*a) - a*Lz;
  float dr = sqrt(max(P*P - D*(Q + (Lz - a*E)*(Lz - a*E)), 0.0)) / S;
  float dth = sqrt(max(Q - cos(th)*cos(th)*(Lz*Lz/(sin(th)*sin(th)) - a*a*E*E), 0.0)) / S;
  float dphi = (Lz/(sin(th)*sin(th)) - a*E + a*P/D) / S;
  float dt = (a*(Lz - a*E*sin(th)*sin(th)) + (r*r + a*a)*P/D) / S;
  return vec4(dr, dth, dphi, dt);
}
```

```glsl
// Add after Kerr metric functions:
float isco_radius(float a_dimless) {
  float a = a_dimless * M;
  float Z1 = 1.0 + pow(1.0 - a*a/(M*M), 1.0/3.0) * (pow(1.0 + a/M, 1.0/3.0) + pow(1.0 - a/M, 1.0/3.0));
  float Z2 = sqrt(3.0*a*a/(M*M) + Z1*Z1);
  return M * (3.0 + Z2 - sqrt((3.0 - Z1)*(3.0 + Z1 + 2.0*Z2)));
}
```

- [ ] **Step 2: Add Kerr orbital 4-velocity (exact, Bardeen et al.)**

```glsl
// Returns u^t, u^phi for circular prograde equatorial orbit at radius r
vec2 kerr_orbital_4vel(float r, float a_dimless) {
  float a = a_dimless * M;
  float r32 = r*sqrt(r);
  float denom = r32 + a*sqrt(M);
  float Omega = sqrt(M) / denom;  // angular velocity at infinity
  float u_t = denom / sqrt(r*r*r - 3.0*M*r*r + 2.0*a*sqrt(M)*r32);
  float u_phi = Omega * u_t;
  return vec2(u_t, u_phi);
}
```

- [ ] **Step 3: Add 4-vector Doppler factor**

```glsl
// Photon 4-momentum at emission: p_mu = (-E, p_r, p_th, p_phi)
// For null geodesic: p_t = -E, p_phi = Lz, p_r = ±sqrt(R)/Sigma, p_th = ±sqrt(Theta)/Sigma
// Doppler factor D = (p_mu u^mu)_emitter / (p_mu u^mu)_observer
// Observer is static at infinity: u_obs = (1, 0, 0, 0) → p_mu u^mu = -p_t = E = 1
// So D = 1 / (p_mu u^mu)_emitter = 1 / (-E*u^t + Lz*u^phi)
float kerr_doppler(float r, float th, float E, float Lz, float Q, float a_dimless) {
  vec2 u = kerr_orbital_4vel(r, a_dimless);
  float p_t = -E;
  float p_phi = Lz;
  float p_mu_u = p_t * u.x + p_phi * u.y;  // u^r = u^th = 0 for circular orbit
  return 1.0 / max(-p_mu_u, 1e-4);
}
```

```glsl
if (crossed_disk) {

  float alpha = prev_cos_th / (prev_cos_th - cos_th);
  float r_cross = prev_r + alpha * (r - prev_r);

  float r_outer = 12.0;
```

- [ ] **Step 1: Add jet emission function**

```glsl
// Add after disk emission helper:
vec3 jet_emission(float r, float th, float E, float Lz, float Q, float a_dimless) {
  float a = a_dimless * M;
  float r_isco = isco_radius(a_dimless);
  
  // Collimation: Gaussian around spin axis (th=0 and th=pi)
  float cos_th = cos(th);
  float collimation = exp(-(1.0 - abs(cos_th))*(1.0 - abs(cos_th)) / (2.0 * 0.07*0.07));  // ~8° half-angle
  
  // Radial profile: power law from ISCO outward
  float radial = pow(r_isco / r, 2.0);
  
  // Knots: slow-moving internal shocks
  float knots = 1.0 + 0.3 * sin(r * 2.0 + u_time * 0.1);
  
  // Doppler beaming along jet axis (bulk beta ~ 0.95)
  float beta_jet = 0.95;
  float gamma_jet = 1.0 / sqrt(1.0 - beta_jet*beta_jet);
  float cos_psi = abs(cos_th);  // angle between jet axis and line of sight
  float D_jet = 1.0 / (gamma_jet * (1.0 - beta_jet * cos_psi));
  float beam_jet = pow(D_jet, 3.0);  // bolometric
  
  // Counter-jet (opposite direction)
  float D_counter = 1.0 / (gamma_jet * (1.0 + beta_jet * cos_psi));
  float beam_counter = pow(D_counter, 3.0);
  
  // Total beaming (jet + counter-jet, but counter-jet is very dim)
  float beam_total = beam_jet + 0.01 * beam_counter;
  
  // Gravitational redshift at base
  float redshift = (1.0 - M/r_isco) * (1.0 - M/r_isco);
  
  // Emissivity
  float j = u_jetBrightness * collimation * radial * knots * beam_total * redshift;
  
  // Color: high-luma end of palette (indices 5..7)
  vec3 jet_col = palette_color(0.8 + 0.2 * (1.0 - r_isco/r));  // hotter near base
  
  return jet_col * j;
}
```

```python
# rebuild_boot.py
import math
from PIL import Image

W, H = 64, 16
img = Image.new('RGBA', (W, H), (0, 0, 0, 0))
px = img.load()

DARK = (0x1c, 0x1f, 0x2e, 255)   # #1c1f2e
AMBER = (0xff, 0x9a, 0x2e, 255)  # #ff9a2e
WHITE = (0xf5, 0xf7, 0xff, 255)  # #f5f7ff

cx, cy = 8, 8  # center of each 16×16 glyph

# Circle r=6 (40 px)
circle = [(cx+dx, cy+dy) for dy in range(-6,7) for dx in range(-6,7)
          if abs(math.hypot(dx,dy) - 6) < 0.5]

# Base arc (bottom, 10 px)
BASE_ARC = [(6,14),(7,14),(8,14),(9,14),(10,14),(11,13),(12,12),(12,13),(13,11),(13,12)]

def rot(p, k):
    dx, dy = p[0]-cx, p[1]-cy
    for _ in range(k%4):
        dx, dy = dy, -dx
    return (cx+dx, cy+dy)

for frame in range(4):
    ox = frame * 16  # x-offset for this glyph
    k = frame  # arc rotates CW 90° per frame
    
    arc = [rot(p, k) for p in BASE_ARC]
    ring = [p for p in circle if p not in arc]
    
    # Draw ring (AMBER)
    for (x, y) in ring:
        px[x + ox, y] = AMBER
    # Draw arc (WHITE)
    for (x, y) in arc:
        px[x + ox, y] = WHITE
    # Draw disk (DARK, 5×5 at local (6,6))
    for dy in range(5):
        for dx in range(5):
            px[6 + dx + ox, 6 + dy] = DARK

img.save('assets/boot-sheet.png')
print('boot-sheet.png rebuilt:', img.size, 'pixels:', sum(1 for _ in img.getdata() if _[3] > 0))
```
